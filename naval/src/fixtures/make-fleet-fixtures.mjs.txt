// Writes the eight boats added on 2026-11-10 (j70, farr40, soling, folkboat,
// cal40, contessa32, hr40c, dinghy470) from the figures quoted below. Kept as
// .txt so nothing builds or tests it; to regenerate, copy it to
// make-fleet-fixtures.mjs in this folder and run it with node (it imports
// ../doc.js and ../numbers.js). Unlike make-fixtures.mjs.txt, the JSON it
// writes is not edited afterwards: this file is the authority for these eight.
//
// Three things are solved here, with the tool's own model, and then written
// as plain numbers:
//
// - BWL, where a source gives a waterplane figure and none gives BWL: the
//   BWL whose synthetic hull has that waterplane area (ORC's "sink", kg/mm,
//   over the certificate's water density) or that BMt (the Contessa 32's,
//   from the Wolfson report). It is marked estimated: the waterplane's
//   shape is the synthetic hull's.
// - The keel's span, so the overall draft is the published one (the root is
//   at the synthetic hull's underside, which no drawing was measured for).
// - Nothing else.
//
// PINS at the bottom are the model's readings measured when each known
// disagreement was recorded. They are pasted in from the suite's own output
// (NAVAL_PRINT_PINS=1 npx vitest run src/validation-fleet.test.js) and are
// what the ratchets hold; re-measure and paste again when a ratchet trips.

const FT = 0.3048, LB = 0.45359237;
const r4 = (v) => Math.round(v * 1e4) / 1e4;
const r6 = (v) => Math.round(v * 1e6) / 1e6;

const lb = (v) => r6(v * LB);
const RET = "2026-11-10";

const foil = (id, ...keys) => keys.map((k) => `appendages.${id}.${k}`);

const notFetched = (id, what, cite, url, note) => ({ id, what, cite, url, retrieved: RET, fetched: false, note });

const readout = (n, id) => n.sections.flatMap((s) => s.rows).find((r) => r.id === id)?.value;
const numbers = (doc, condition) => {
  const d = JSON.parse(JSON.stringify(doc));
  if (condition) d.loading = { ...d.loading, displacement: { from: "given", kg: condition.displacementKg }, crew: { ...d.loading.crew, ...condition.crew } };
  return computeNumbers(normalize(d));
};

function solve(boat) {
  const { doc, solve: s, published } = boat;
  const cond = (name) => published.conditions?.[name];
  const bisect = (lo, hi, f) => {
    for (let i = 0; i < 40; i++) {
      const mid = 0.5 * (lo + hi);
      if (f(mid) < 0) lo = mid;
      else hi = mid;
    }
    return r4(0.5 * (lo + hi));
  };
  const withBwl = (b) => ((doc.hull.dims.bwl = b), doc);
  if (s.awM2) doc.hull.dims.bwl = bisect(0.4 * doc.hull.dims.beam, doc.hull.dims.beam, (b) => numbers(withBwl(b), cond(s.awCondition)).form.hydro.awM2 - s.awM2);
  if (s.bmtM) doc.hull.dims.bwl = bisect(0.4 * doc.hull.dims.beam, doc.hull.dims.beam, (b) => numbers(withBwl(b), cond(s.bmCondition)).form.hydro.bmt - s.bmtM);
  for (let i = 0; i < 8; i++) {
    const d = readout(numbers(doc), "draftOverall");
    doc.appendages.keel.span = r4(doc.appendages.keel.span + (s.draft - d));
  }
}
